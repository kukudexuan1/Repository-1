    if St == float('inf'):

## 3 齿轮传动

### calc#13 齿轮安装段挠度与转角校核（阶梯轴单位载荷法）

依据：阶梯轴弯曲刚度：单位载荷法（虚功法）数值积分，y=∫M(x)m(x)/(EI(x))dx、θ=∫M(x)m_θ(x)/(EI(x))dx；EI(x) 按实际阶梯截面分段（φ35/φ40/φ45及键槽削弱段）；载荷分两平面（Ft 平面、Fr+Fa 力偶平面）分别计算后矢量合成。材料力学（刘鸿文）单位载荷法。

公式：`（沙箱脚本，见 script 字段）`

输入：E = 2.06e+05 MPa，L = 202 mm，a = 108 mm

代入：E=2.06e+05 MPa；L=202 mm；a=108 mm

结果：y_108 = 0.01892 mm，theta_80 = 8.865e-05 rad，theta_108 = 3.622e-05 rad，theta_136 = 0.0001461 rad

计算脚本：

```python
E = 206000.0
L = 202.0
a = 108.0
b = 94.0
Ft = 3200.0
Fr = 1300.0
M_Fa = 27562.5

R_A1 = Ft*b/L   # 1489.109
R_A2 = Fr*b/L   # 604.950
R_Ac = M_Fa/L   # 136.448

I35 = 73661.75743426848
I40 = 125663.70614359173
I45 = 201288.95898635383
I45k = 171254.14648635383

def I_x(x):
    if x < 10.5:   return I35
    elif x < 80.0: return I40
    elif x < 83.0: return I45
    elif x < 133.0:return I45k
    elif x < 136.0:return I45
    elif x < 191.5:return I40
    else:          return I35

def M1(x):
    return R_A1*x if x <= a else R_A1*x - Ft*(x-a)

def M2(x):
    if x <= a:
        mfr = R_A2*x
        mc = R_Ac*x
    else:
        mfr = R_A2*x - Fr*(x-a)
        mc = R_Ac*x - M_Fa
    return mfr + mc

def m_unit(x, x0):
    rA = (L - x0)/L
    return rA*x if x <= x0 else rA*x - (x - x0)

def m_couple(x, x0):
    rA = 1.0/L
    return rA*x if x < x0 else rA*x - 1.0

def integrate(f, lo, hi, n=4000):
    dx = (hi-lo)/n
    s = 0.0
    for i in range(n+1):
        x = lo + i*dx
        w = 1.0 if (i==0 or i==n) else (4.0 if i%2==1 else 2.0)
        s += w*f(x)
    return s*dx/3.0

def deflection(x0):
    y1 = integrate(lambda x: M1(x)*m_unit(x,x0)/(E*I_x(x)), 0.0, L)
    y2 = integrate(lambda x: M2(x)*m_unit(x,x0)/(E*I_x(x)), 0.0, L)
    return math.hypot(y1, y2)

def slope(x0):
    t1 = integrate(lambda x: M1(x)*m_couple(x,x0)/(E*I_x(x)), 0.0, L)
    t2 = integrate(lambda x: M2(x)*m_couple(x,x0)/(E*I_x(x)), 0.0, L)
    return math.hypot(t1, t2)

y_108 = deflection(108.0)
theta_80 = slope(80.0)
theta_108 = slope(108.0)
theta_136 = slope(136.0)

outputs = {
    'y_108': y_108,
    'theta_80': theta_80,
    'theta_108': theta_108,
    'theta_136': theta_136,
}
```
